   {prepared&&<><div className="review-summary"><span>{prepared.cashDelta<0?'USDC paid':prepared.cashDelta>0?'USDC received':'NVDAx deposited'}</span><strong>{prepared.cashDelta!==0?usd(Math.abs(prepared.cashDelta)):units(Math.abs(prepared.stockDelta))}</strong></div><dl className="review-rows"><div><dt>Wallet</dt><dd>{short(prepared.wallet)}</dd></div><div><dt>NVDAx change</dt><dd>{prepared.stockDelta>=0?'+':'−'}{units(Math.abs(prepared.stockDelta))}</dd></div><div><dt>Network fee</dt><dd>{prepared.networkFee.toFixed(6)} SOL</dd></div><div><dt>Total SOL debit incl. setup</dt><dd>{prepared.solCost.toFixed(6)} SOL</dd></div>{prepared.kind==='sale'&&<div><dt>Minimum USDC received</dt><dd>{usd(prepared.minimumCash)}</dd></div>}<div><dt>Simulation slot</dt><dd>{prepared.simulationSlot.toLocaleString()}</dd></div></dl>
   {prepared.kind==='deposit'&&<p className="review-note">This deposits collateral only. After confirmation, refresh and review borrowing as a separate transaction. If borrowing cannot proceed, use Release collateral to request withdrawal.</p>}
   {!receipt&&!sent&&<><p className="review-note">{clock>=prepared.expiresAt?'Review expired. Prepare a fresh transaction.':'This moves real assets. Simulation passed; your wallet has not signed or sent it.'}</p><Button disabled={clock>=prepared.expiresAt||signing||prepared.wallet!==address} onClick={()=>void send()}>Sign {prepared.kind==='sale'?'sale':prepared.kind} in wallet <ArrowRight/></Button>{clock>=prepared.expiresAt&&<Button variant="outline" onClick={()=>void prepare(prepared.kind)}>Prepare again</Button>}</>}
   {signing&&<p role="status">Awaiting your wallet’s decision…</p>}
   {receipt&&<div className="transaction-status" role="status"><strong>{successful?'Confirmed':receipt.status==='failed'?'Transaction failed':'Waiting for network confirmation'}</strong><a href={`https://solscan.io/tx/${receipt.signature}`} target="_blank" rel="noreferrer">View transaction <ExternalLink size={13}/></a>{successful&&<><p>Actual USDC change: {usd(Number(receipt.cashRawDelta)/1e6)}</p><Button variant="outline" onClick={()=>download('stockroom-receipt.json',receipt)}><Download size={15}/>Download receipt</Button><Button onClick={()=>{setAction(null);setRefresh(v=>v+1)}}>Refresh wallet & position</Button></>}</div>}
   {receiptError&&<p className="wallet-error">{receiptError}</p>}
   {!sent&&<Button variant="ghost" onClick={()=>download('stockroom-unsigned-review.json',prepared)}><Download size={14}/>Download unsigned review</Button>}
   </>}
   {!preparing&&!prepared&&action&&<Button variant="outline" onClick={()=>void prepare(action)}>Check again</Button>}
  </DialogContent></Dialog>
 </main>
